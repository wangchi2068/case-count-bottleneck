\section{Background and theoretical framework}
\label{sec:background}

\subsection{Transmission, reporting, and the two estimation targets}
\label{sec:bg-model}
\label{sec:meth-model}
\label{sec:meth-targets}

Consider transmission during discrete generation period $t$. Following \citet{lloydsmith2005superspreading}, we model superspreading and individual-level transmission heterogeneity by assuming that, conditional on the instantaneous reproduction number $R_t$, each of the $n$ primary infected individuals generates a random number of secondary infections $X_{i,t}$ drawn independently from a negative-binomial offspring distribution:
\begin{equation}
  X_{i,t}\mid R_t \sim \NB\!\left(k,\ \frac{k}{k+R_t}\right),\qquad
  \E(X_{i,t}\mid R_t)=R_t,\quad \Var(X_{i,t}\mid R_t)=R_t+\frac{R_t^2}{k},
  \label{eq:offspring}
\end{equation}
where $k > 0$ is the overdispersion parameter; smaller values of $k$ denote heavier overdispersion, and $k \to \infty$ recovers Poisson transmission. The total incidence of secondary infections in the generation is $I_t = \sum_{i=1}^n X_{i,t}$. By infinite divisibility and the convolution property of the negative-binomial family, conditional on $R_t$ and $n$, total infections follow
\begin{equation}
  I_t\mid n, R_t \sim \NB\!\left(nk,\ \frac{k}{k+R_t}\right),\qquad
  \E(I_t\mid n, R_t)=nR_t,\quad \Var(I_t\mid n, R_t)=nR_t+\frac{nR_t^2}{k}.
  \label{eq:incidence}
\end{equation}

Surveillance typically captures only a fraction of total infections. We model case observation through binomial thinning: conditional on total infections $I_t$, each infection is independently confirmed and reported with probability $\rho \in (0, 1]$, yielding the observed case count $C_t \sim \Bin(I_t, \rho)$. A well-known preservation property of the negative-binomial distribution under binomial thinning ensures that the reported count retains an exact negative-binomial form:
\begin{equation}
  C_t\mid n, R_t \sim \NB\!\left(nk,\ \frac{k}{k+\rho R_t}\right),\qquad
  \E(C_t\mid n, R_t)=\rho n R_t,\quad
  \Var(C_t\mid n, R_t)=\rho n R_t+\frac{\rho^2 n R_t^2}{k}.
  \label{eq:observed}
\end{equation}
We define $m = \rho n$ as the expected reporting volume, which serves as the natural scale parameter for the surveillance system.

We formulate the analysis under the following foundational assumptions:
\begin{description}
  \item[H1 (Binomial observation):] Conditional on $I_t$, each infection is ascertained independently with a time-invariant reporting probability $\rho \in (0, 1]$.
  \item[H2 (Homogeneous branching):] Conditional on $R_t$, primary infections generate secondary infections independently and identically according to Eq.~\eqref{eq:offspring}.
  \item[H3 (Exogenous overdispersion):] The dispersion parameter $k$ is an invariant characteristic of the pathogen and host population, independent of $R_t$ and $n$.
  \item[H4 (Predetermined ancestor scale):] The active ancestor volume $n$ is fixed by study design (e.g., in active contact tracing) and is statistically independent of $R_t$.
\end{description}

The actual transmission potential $R_t$ fluctuates over time. Meteorology, school vacations, public holidays, and behavioral shifts drive continuous variation around a baseline level \citep{dalziel2018urbanization, rasmussen2011inference}. We denote the marginal expectation and environmental variance by
\begin{equation}
  \E(R_t) = \Rbar, \qquad \Var(R_t) = v_R.
\end{equation}
In ecological population biology, fluctuations shared by all co-circulating lineages within a time interval are termed environmental stochasticity, distinguishing them from individual-level demographic stochasticity \citep{engen1998demographic, rohani2002determinism}.

This distinction defines two fundamentally different estimation targets:
\begin{itemize}
  \item \textbf{Target 1 (Conditional instantaneous reproduction number, $R_t$):} The realised transmission potential in the current period given the ambient environment. This target is relevant for real-time situational awareness, outbreak detection, and evaluating immediate intervention efficacy.
  \item \textbf{Target 2 (Marginal cross-period mean, $\Rbar$):} The central tendency of transmission potential across periods. Under temporal independence of environmental conditions, $\Rbar$ also constitutes the minimum mean squared error one-step-ahead forecast for the next generation.
\end{itemize}

Both targets share the natural plug-in estimator $\widehat{R}_t = C_t / m = C_t / (\rho n)$. However, their estimation errors differ fundamentally in their reference benchmark: the error for Target 1 is $\widehat{R}_t - R_t$, whereas the error for Target 2 is $\widehat{R}_t - \Rbar = (\widehat{R}_t - R_t) + (R_t - \Rbar)$, which carries an additional, sample-invariant environmental fluctuation term $R_t - \Rbar$.

\begin{proposition}[Estimation of Target 1]
\label{prop:current}
Under H1--H4, for any given ancestor scale $n$ and reporting probability $\rho$, the estimator $\widehat{R}_t = C_t / m$ satisfies:
\begin{enumerate}[label=(\roman*)]
  \item It is conditionally unbiased for $R_t$: $\E(\widehat{R}_t \mid R_t) = R_t$.
  \item Its conditional variance decays strictly as $1/m$:
  \begin{equation}
    \Var(\widehat{R}_t\mid R_t) = \frac{R_t}{\rho n} + \frac{R_t^2}{kn} = \frac{R_t}{m} + \frac{\rho R_t^2}{km}.
    \label{eq:var-current}
  \end{equation}
  \item It attains the Cram\'er--Rao lower bound: the conditional Fisher information for $R_t$ is $\mathcal{I}(R_t) = [R_t/m + \rho R_t^2/(km)]^{-1}$, and $\widehat{R}_t$ is the unique uniformly minimum-variance unbiased estimator (UMVUE).
\end{enumerate}
\end{proposition}

Proposition~\ref{prop:current} confirms that for Target 1, increasing sample volume $m$ continuously and monotonically suppresses estimation variance toward zero, consistent with classical information-theoretic scaling \citep{parag2022quantifying}.

\subsection{Variance decomposition and the balance scale}
\label{sec:bg-variance}

For Target 2, the behavior of $\widehat{R}_t$ departs sharply from the $1/m$ scaling law.

\begin{proposition}[Estimation variance and balance scale for Target 2]
\label{prop:average}
Under H1--H4, the marginal variance of $\widehat{R}_t$ around $\Rbar$ decomposes into:
\begin{equation}
  V(m) = \Var(\widehat{R}_t \mid n) = v_R + \frac{\Rbar}{\rho n} + \frac{\Rbar^2 + v_R}{kn} = v_R + \frac{A}{m},
  \label{eq:var-average}
\end{equation}
where the scale-dependent coefficient $A$ is
\begin{equation}
  A = \Rbar + \frac{\rho(\Rbar^2 + v_R)}{k}.
  \label{eq:A-def}
\end{equation}
Defining the variance balance scale $m_\times = A / v_R$, the error variance satisfies:
\begin{enumerate}[label=(\roman*)]
  \item \textbf{Asymptotic floor:} As $m \to \infty$, $V(m) \to v_R > 0$.
  \item \textbf{Variance partition:} The sampling share of total variance is $1 / (1 + m/m_\times)$; at $m = m_\times$, the sampling variance exactly equals the environmental variance floor.
  \item \textbf{Diminishing returns to scale doubling:} Doubling sample volume from $m$ to $2m$ reduces total variance by a fraction of $1 / [2(1 + m/m_\times)]$.
  \item \textbf{Elasticity:} $V(m)$ is strictly decreasing and strictly convex, with scale elasticity $\epsilon_m = -m_\times / (m + m_\times)$, which equals $-1/2$ at $m = m_\times$.
  \item \textbf{Optimality bound:} In the class of estimators conditionally unbiased for $R_t$, $V(m)$ is the minimum mean squared error for $\Rbar$. Under H4 with $R_{t+1}$ independent of past history $\mathcal{F}_t$, it is also the minimum mean squared prediction error for one-step-ahead growth $y = C_{t+1}/m$.
\end{enumerate}
\end{proposition}

We emphasize that the balance scale $m_\times = A/v_R$ serves solely as a comparative benchmark where sampling and environmental variances are of equal magnitude; it does not prescribe a minimum operational threshold satisfying specific estimation precision or outbreak detection power requirements.

Equation~\eqref{eq:var-average} follows directly from the law of total variance: the expected conditional variance yields $A/m = \E[1/\mathcal{I}(R_t)]$, whereas the variance of the conditional expectation yields $v_R$. For any estimator $\tilde{\theta}$ conditionally unbiased for $R_t$, the cross-term in $\E[(\tilde{\theta}-R_t)(R_t-\Rbar)]$ vanishes identically:
\begin{equation}
  \E(\tilde{\theta} - \Rbar)^2 = \E\big[\Var(\tilde{\theta}\mid R_t)\big] + v_R \ge v_R + \E\big[1/\mathcal{I}(R_t)\big] = v_R + \frac{A}{m},
  \label{eq:unbiased-bound}
\end{equation}
with equality attained by $\widehat{R}_t = C_t/m$. For the one-step prediction result, we additionally assume that $R_{t+1}$ is independent of $\mathcal{F}_t$. Under this condition, any predictor $\hat{y}$ based on history $\mathcal{F}_t$ forecasting $y = C_{t+1}/m$ satisfies
\begin{equation}
  \E(\hat{y} - y)^2 \ge \Var(y \mid \mathcal{F}_t) = v_R + \frac{A}{m},
  \label{eq:predict-bound}
\end{equation}
with equality attained by $\hat{y} = \Rbar$.

Because $V(m)$ is strictly convex with no inflection points, $m_\times$ provides a natural dimensional reference (Fig.~\ref{fig:variance}B): when $m \ll m_\times$, error is dominated by demographic and reporting sampling noise; when $m \gg m_\times$, error is dominated by environmental stochasticity, and further sample expansion faces rapidly diminishing returns. For typical baseline parameters ($\Rbar=1.15$, $v_R=0.04$, $k=0.35$, $\rho=0.25$), we have $A = 2.1232$ and $m_\times = 53.1$ cases. At sample volumes of $m = 10$, $m_\times = 53.1$, and $m = 200$, doubling sample size reduces total variance by 42.1\%, 25.0\%, and 10.5\%, respectively (Fig.~\ref{fig:variance}C).

\begin{remark}[Reporting completeness versus population coverage]
\label{rem:reporting}
Expected reporting volume $m = \rho n$ can be increased either by raising reporting completeness $\rho$ or by expanding population coverage $n$. Holding $n$ fixed and driving $\rho \to 1$ yields an asymptotic variance limit of $v_R + \Rbar/n + (\Rbar^2+v_R)/(kn)$, which remains strictly greater than $v_R$. Conversely, holding $\rho$ fixed and expanding $n \to \infty$ drives variance to the theoretical floor $v_R$. Increasing $n$ by a small percentage yields a variance reduction that is $A/\Rbar = 1 + \rho(\Rbar^2+v_R)/(k\Rbar)$ times larger than achieving the same percentage gain via $\rho$.
\end{remark}

The lower bound $v_R$ applies strictly to single-period, conditionally unbiased estimators under temporally independent environments ($\phi = 0$). When $R_t$ follows a stationary AR(1) process with autocorrelation $\phi$, historical information can be borrowed, reducing the irreducible one-step-ahead environmental component to $(1-\phi^2)v_R$ (yielding $0.75v_R$ for $\phi=0.5$ and $0.36v_R$ for $\phi=0.8$). Similarly, pooling $J$ jurisdictions with identical environmental variance $v_R$ and average pairwise cross-jurisdiction correlation $\bar{c}$ reduces the environmental variance to
\begin{equation}
  v_R\,\bar{c} + \frac{v_R(1 - \bar{c})}{J}.
  \label{eq:pooling}
\end{equation}
When local disturbances are imperfectly correlated ($\bar{c} < 1$), spatial pooling systematically compresses the environmental floor; when shocks are perfectly synchronised ($\bar{c} = 1$), spatial aggregation provides no reduction.

\subsection{Reporting heterogeneity and identifiability}
\label{sec:bg-reporting}

When reporting probability fluctuates across time, an additional noise component enters the error floor. We replace H1 with:
\begin{description}
  \item[H5 (Stochastic reporting):] The reporting probability $p_t \in (0, 1]$ is a stationary stochastic process independent of $R_t$ and $n$, with $\E(p_t) = \rho$, $\Var(p_t) = \sigma^2$, relative volatility $\tau = \sigma/\rho$, and standardised rate $u_t = p_t/\rho$.
  \item[H6 (Conditional independence):] Given trajectories $(R_1, \dots, R_T)$ and $(p_1, \dots, p_T)$, observed counts across periods are mutually independent.
\end{description}

\begin{proposition}[Variance decomposition under reporting heterogeneity]
\label{prop:report-noise}
Under H2--H6, denoting $U = (1+\tau^2)(\Rbar^2 + v_R)$:
\begin{enumerate}[label=(\roman*)]
  \item The marginal variance of $\widehat{R}_t = C_t / m$ decomposes into three components:
  \begin{equation}
    V_H(m) = \Var(\widehat{R}_t \mid n) = v_R + v_H + \frac{A_H}{m},
    \label{eq:r3-vH}
  \end{equation}
  where the reporting heterogeneity variance $v_H$ and modified scale coefficient $A_H$ are
  \begin{equation}
    v_H = \tau^2(\Rbar^2 + v_R), \qquad A_H = \Rbar + \frac{\rho U}{k}.
    \label{eq:vH-AH}
  \end{equation}
  \item The generalised balance scale becomes $m_{\times, H} = A_H / (v_R + v_H) < m_\times$.
  \item As $m \to \infty$, $V_H(m) \to v_R + v_H$.
  \item \textbf{Unidentifiability:} From the marginal mean and variance of the counts alone, environmental variance $v_R$ and reporting heterogeneity $v_H$ are not separately identifiable; only the compound parameter $U = (1+\tau^2)(\Rbar^2+v_R)$ is identified from these marginal moments.
  \item \textbf{Reporting signature:} When evaluating growth ratios $G_t = C_t / D_t$ with denominator $D_t = \sum_s w_s C_{t-s}$ under stationary AR(1) reporting shocks with persistence $\psi$, the prediction error intercept depends on lag weights via
  \begin{equation}
    F(w, \psi) = 1 - 2\sum_s w_s \psi^s + \sum_s \sum_{s'} w_s w_{s'} \psi^{|s - s'|},
    \label{eq:r3-Fw}
  \end{equation}
  generating diagnostic intercept ratios that differentiate reporting noise from transmission dynamics.
\end{enumerate}
\end{proposition}

Under baseline parameters with $\tau = 0.15$, reporting heterogeneity introduces $v_H = 0.0307$ and compresses the balance scale from $m_\times = 53.1$ down to $m_{\times, H} = 30.4$ (Fig.~\ref{fig:variance}D).

\begin{corollary}[Ranking reversal between reporting completeness and coverage]
\label{cor:switch}
When deciding whether to allocate resources to expanding population coverage $n$ or raising reporting completeness $\rho$:
\begin{enumerate}[label=(\roman*)]
  \item If reporting fluctuations are institutional and scale-invariant ($\mathrm{d}\tau^2/\mathrm{d}\rho = 0$), expanding coverage $n$ strictly dominates increasing $\rho$.
  \item If reporting fluctuations stem from binomial sampling where $\Var(p_t) = \rho(1-\rho)/(\nu+1)$ with sample size $\nu$, the ranking reverses: raising $\rho$ strictly dominates expanding $n$ when
  \begin{equation}
    -\rho\,\frac{\mathrm{d}\tau^2}{\mathrm{d}\rho}\,(kn + 1) > 1 + \tau^2 \quad \iff \quad m > \frac{\rho^2 \nu}{k}.
    \label{eq:r3-switch}
  \end{equation}
\end{enumerate}
\end{corollary}

For $\nu = 100$ and $k = 0.35$, the reversal occurs at $m > 17.9$ cases; for $\nu = 1000$, it occurs at $m > 178.6$ cases (Fig.~\ref{fig:variance}E).
